\begin{frame}
    \frametitle{Syntax}
    Expressions, conditions and statements are defined inductively.
    Let \(n\) be numeric literals and \(v\) variables.

    \[
        \begin{array}{rcl}
            e & := & n
                \; \mid \; v
                \; \mid \; e + e
                \; \mid \; e - e
                \; \mid \; e \times e
                \; \mid \; e \div e
                \\
            c & := & \true
                \; \mid \; \false
                \; \mid \; e \leq e
                \; \mid \; e = e
                \; \mid \; c \land c
                \; \mid \; c \lor c
                \; \mid \; \lnot c
                \\
            s & := & v \gets e
                \; \mid \; s; s
                \; \mid \; \ite{c}{s}{s} \\
            & & \; \mid \; \while{c}{s}
                \; \mid \; \mathbf{explode}
                \; \mid \; \mathbf{skip}
        \end{array}
    \]
\end{frame}

\begin{frame}
    \frametitle{Order theory 1}
    \begin{definition}
        A poset is a set \(P\) equipped with a partial order \(\sqsubseteq\) that's reflexive, antisymmetric and transitive.
    \end{definition}
    (I really hope you've seen this before else this talk will go way too fast)
    \uncover<2->{
    \begin{example}
        For any set \(S\), \((\mathcal{P}(S), \subseteq)\) is a poset.
    \end{example}
    }
\end{frame}

\begin{frame}
    \frametitle{Order theory 1}
    \begin{definition}
        A map \(f : P \to Q\) between posets is monotone if whenever \(x \sqsubseteq_P y\), \(f(x) \sqsubseteq_Q f(y)\).
    \end{definition}
    \uncover<2->{
    \begin{proposition}
        The identity map is monotone. Composites of monotone maps are monotone.
    \end{proposition}
    \begin{proof}
        If \(x \sqsubseteq y\) then \(x \sqsubseteq y\).

        If \(x \sqsubseteq y\) then \(f(x) \sqsubseteq f(y)\) then \(g(f(x)) \sqsubseteq g(f(y))\).
    \end{proof}
    }
    \uncover<3->{
    Monotone maps are ordered pointwise, \(f \sqsubseteq g\) iff \(f(x) \sqsubseteq g(x)\) for all \(x\) in the domain.
    }
\end{frame}

\begin{frame}
    \frametitle{Approximating semantics}
    Instead of defining the full semantics, define an approximation set of states \(\statesetb\) and (letting \(\concdomainb = \mathcal{P}(\statesetb)\)) define an approximation operator \(\statementb{s} : \concdomainb \to \concdomainb\) for each statement \(s\).

    \uncover<2->{
    Need a monotone map \(\hat{\gamma} : \concdomainb \to \concdomain\) such that \(\statement{s} \circ \gamma \sqsubseteq \gamma \circ \statementb{s}\).
    \[
    \begin{tikzcd}[ampersand replacement=\&]
        \concdomainb \arrow[r, "\statementb{s}"] \arrow[d, "\gamma", swap] \& \concdomainb \arrow[d, "\gamma"] \\
        \concdomain \arrow[r, "\statement{s}", swap] \arrow[ur, "\sqsubseteq" {rotate=45}, phantom] \& \concdomain
    \end{tikzcd}
    \]
    }
    
    \uncover<3->{
    If you prefer, we need \(\hat{\gamma}\) a lax natural transformation viewing semantics opeators as functors from the one object category of statements under sequential composition to the poset-enriched category of posets!
    }
\end{frame}

\begin{frame}
    \frametitle{Approximating semantics}
    Define \(\statesetb = \ZZ^V\) (no arrays!) and \(\concdomainb = \mathcal{P}(\statesetb)\).
    Then define \(\gamma : \concdomainb \to \concdomain\) by \(\gamma(S) = S \times A^W\), giving all possible assignments of array variables.
    This is clearly monotone!
\end{frame}
